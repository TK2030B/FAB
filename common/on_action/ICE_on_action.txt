on_actions = {

	on_daily_ICE = {
		effect = {
			if = {
				limit = { is_in_faction = yes }
				leave_faction = yes
			}
			every_country = {
				limit = {
					exists = yes
					NOT = { tag = ICE }
					has_non_aggression_pact_with = ICE
				}
				diplomatic_relation = {
					country = ICE
					relation = non_aggression_pact
					active = no
				}
			}
		}
	}


}
