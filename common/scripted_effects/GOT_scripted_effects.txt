        add_stability = -0.01
    }

GOT_Sparrows_Angered = {

# No previous relationship
	if = {
		limit = {
			NOT = { has_idea = CRO_Sparrows_Angry }
			NOT = { has_idea = CRO_Sparrows_Angry_2 }
			NOT = { has_idea = CRO_Sparrows_Angry_3 }
			NOT = { has_idea = CRO_Sparrows_Placated }
			NOT = { has_idea = CRO_Sparrows_Placated_2 }
			NOT = { has_idea = CRO_Sparrows_Placated_3 }
		}

		add_ideas = CRO_Sparrows_Angry
	}

	# From Angry -> Angry 2
	else_if = {
		limit = {
			has_idea = CRO_Sparrows_Angry
		}

		remove_ideas = CRO_Sparrows_Angry
		add_ideas = CRO_Sparrows_Angry_2
	}

	# From Angry 2 -> Angry 3
	else_if = {
		limit = {
			has_idea = CRO_Sparrows_Angry_2
		}

		remove_ideas = CRO_Sparrows_Angry_2
		add_ideas = CRO_Sparrows_Angry_3
	}

	# If already Angry 3, nothing happens
	
	else_if = {
		limit = {
			has_idea = CRO_Sparrows_Placated_3
		}

		remove_ideas = CRO_Sparrows_Placated_3
		add_ideas = CRO_Sparrows_Angry
	}

	else_if = {
		limit = {
			has_idea = CRO_Sparrows_Placated_2
		}

		remove_ideas = CRO_Sparrows_Placated_2
		add_ideas = CRO_Sparrows_Angry
	}

	else_if = {
		limit = {
			has_idea = CRO_Sparrows_Placated
		}

		remove_ideas = CRO_Sparrows_Placated
		add_ideas = CRO_Sparrows_Angry
	}

}

	if = {

		add_political_power = 100
		add_stability = 0.10

	# Faith was a major rival - Crown restores authority
	else_if = {
		limit = { has_idea = CRO_Sparrows_Angry_2 }
		add_political_power = 75
		add_stability = 0.075
	}

	# Some tension, but victory is welcomed
	else_if = {
		limit = { has_idea = CRO_Sparrows_Angry }
		add_political_power = 40
		add_stability = 0.05
	}

	# Crown breaks an agreement with the Faith
	else_if = {
		limit = { has_idea = CRO_Sparrows_Placated }
		add_political_power = -40
		add_stability = -0.05
	}

	# Crown empowered the Faith, then betrayed them
	else_if = {
		limit = { has_idea = CRO_Sparrows_Placated_2 }
		add_political_power = -75
		add_stability = -0.10
	}

	# The Crown created this problem and then destroyed its own ally
	else_if = {
		limit = { has_idea = CRO_Sparrows_Placated_3 }
		add_political_power = -100
		add_stability = -0.15
	}

	GOT_End_of_Sparrows = yes

		limit = {

		FAI = {
			load_oob = FAI_Large
		}
		
	}

	# Faith was not empowered - smaller uprising
	else = {

		add_stability = -0.10
		add_political_power = -50

		FAI = {
			load_oob = FAI_Small
		}
	}
	
	GOT_End_of_Sparrows = yes
}
