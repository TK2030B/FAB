GOT_handle_factories = {
	GOT_handle_factories_westeros = yes
	GOT_handle_factories_essos = yes
}

GOT_handle_factories_westeros = {
		#Crownlands
		if = {
			limit = {
				tag = CRO
			}
			#19 Dockyards
			for_loop_effect = {
                start = 1
                end = 19
                compare = less_than_or_equals
                random_coastal_state = {
                    limit = {
                        is_owned_and_controlled_by = PREV
                    }
                    add_extra_state_shared_building_slots = 1
                    add_building_construction = {
                        type = dockyard
                        level = 1
                        instant = yes
                    }
                }
            }
			#15 Military Factories
			for_loop_effect = {
				start = 1
				end = 15
				compare = less_than_or_equals
				random_owned_controlled_state = {
					limit = {
						free_building_slots = {
							building = arms_factory
							size > 0
						}
					}
					add_extra_state_shared_building_slots = 1
					add_building_construction = {
						type = arms_factory
						level = 1
						instant = yes
					}
				}
			}
			#32 Civilian Factories
			for_loop_effect = {
				start = 1
				end = 32
				compare = less_than_or_equals
				random_owned_controlled_state = {
					limit = {
						free_building_slots = {
							building = industrial_complex
							size > 0
						}
					}
					add_extra_state_shared_building_slots = 1
					add_building_construction = {
						type = industrial_complex
						level = 1
						instant = yes
					}
				}
			}
		}
		#Paste New Countries before this line
}
