#2. Stannis Baratheon Declares His Claim to the Throne
country_event = {
    id = GOT_Story_Country.101
    title = GOT_Story_Country.101.t
    desc = GOT_Story_Country.101.d
    picture = GFX_report_event_GOT_Story_Country101

    trigger = {
        tag = DRA
		date > 1937.1.5
        date < 1937.2.5
		country_exists = CRO
		CRO = {
			is_subject = no
		}
		is_subject = no
		#Obviously not Robert or Himself on the throne for him to make this decision
		NOT = {
			CRO = {
				has_country_leader = {
					name = "Robert Baratheon"
					ruling_only = yes
				}
			}
			CRO = {
				has_country_leader = {
					name = "Stannis Baratheon"
					ruling_only = yes
				}
			}
		}
    }

    fire_only_once = yes

    option = {
        name = GOT_Story_Country.101.a

        ai_chance = {
            factor = 70
			 modifier = {
				factor = 1.35    # Boosts base 70 to 95 (70 * 1.35 = 94.5)
				has_global_flag = GOT_Lord_Stark_Investigation_Flag
			}
			modifier = {
				factor = 0.71    # Reduces base 70 to ~50 (70 * 0.71 ≈ 50)
				NOT = { has_global_flag = GOT_Lord_Stark_Investigation_Flag }
			}
        }
		
		add_threat = 10
        add_stability = -0.05
        add_political_power = -50
        add_war_support = 0.05
		
		set_country_flag = GOT_Claimed_Iron_Throne_Flag
		
		#If Dragonstone is in a faction with the Crownlands, then he leaves
		if = {
			limit = {
				is_in_faction_with = CRO
			}
			leave_faction = yes
		}
		
		create_wargoal = {
			type = annex_everything
			target = CRO
			expire = 365
		}
		#Every other country informed of Stannis claim
        every_other_country = {
            limit = {
                exists = yes
            }
            news_event = {
                id = GOT_Story_News.101
                hours = 6
            }
        }
    }
	
	option = {
        name = GOT_Story_Country.101.b
		
		trigger = {
			NOT = {
				has_war_with = CRO
			}
		}

        ai_chance = {
            factor = 25
			modifier = {
				factor = 0.2   # 25 * 0.2 = 5
				has_global_flag = GOT_Lord_Stark_Investigation_Flag
			}
			modifier = {
				factor = 2.0   # 25 * 2.0 = 50
				NOT = { has_global_flag = GOT_Lord_Stark_Investigation_Flag }
			}
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
        }

        add_stability = 0.05
        add_political_power = 25
		
		set_country_flag = GOT_Bent_The_Knee_Flag
		
		#If Dragonstone is a faction leader, the faction is disbaned
		if = {
			limit = {
				is_faction_leader = yes
			}
			disband_faction = yes
		}

		#If Dragonstone is a faction member, but not the leader, then he just leaves
		if = {
			limit = {
				is_in_faction = yes
				NOT = { is_in_faction_with = CRO }
			}
			leave_faction = yes
			#Crownlands gets event asking them to join their faction
			every_other_country = {
				limit = {
					is_in_faction_with = CRO
					is_faction_leader = yes
				}
			   country_event = {
					id = generic.2
					hours = 6
				}
			}
		}
		
		#Everyone gets the generic Bend the Knee Event
        every_other_country = {
            limit = {
                exists = yes
            }
            news_event = {
                id = GOT_Diplomatic_News.9
                hours = 6
            }
        }
    }
	
	# Option to bend the knee to Renly if he's declared himself king
    option = {
        name = GOT_Story_Country.101.c
		
		ai_chance = {
            factor = 5
			modifier = {
				factor = 0.5   # 5 * 0.5 = 2.5 → investigation makes this even less likely
				has_global_flag = GOT_Lord_Stark_Investigation_Flag
			}
			modifier = {
				factor = 2.0   # 5 * 2.0 = 10 → without investigation, slightly more plausible
				NOT = { has_global_flag = GOT_Lord_Stark_Investigation_Flag }
			}
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
        }

        trigger = {
			NOT = {
				has_war_with = STO
			}
			OR = {
				STO = {
					has_country_flag = GOT_Claimed_Iron_Throne_Flag
				}
				CRO = {
					has_country_leader = {
						name = "Renly Baratheon"
						ruling_only = yes
					}
				}
			}
        }

        add_stability = 0.10
        add_political_power = 50

        set_country_flag = GOT_Bent_The_Knee_Flag
		
		#If Dragonstone is a faction leader, the faction is disbaned
		if = {
			limit = {
				is_faction_leader = yes
			}
			disband_faction = yes
		}

		#If Dragonstone is a faction member, but not the leader, then he just leaves
		if = {
			limit = {
				is_in_faction = yes
				NOT = { is_in_faction_with = STO }
			}
			leave_faction = yes
			#Stormlands gets event asking them to join their faction
			STO = {
				country_event = {
					id = generic.2
					hours = 6
				}
			}
		}

        # Inform the world that Stannis bent the knee to Renly
        every_other_country = {
            limit = {
                exists = yes
            }
            news_event = {
                id = GOT_Diplomatic_News.12
                hours = 6
            }
        }
    }
}
