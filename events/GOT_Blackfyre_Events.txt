#Blackfyre Spawn Volantis
country_event = {
	id = GOT_Blackfyre_Country.2
	title = GOT_Blackfyre_Country.2.t
	desc = GOT_Blackfyre_Country.2.d
	picture = GFX_report_event_blackfyre_spawn

	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			random_owned_state = {
				limit = {
					is_owned_by = VOL
					is_capital = no
					is_island_state = no
				}
				# Transfer ownership and control to the target tag
				transfer_state_to = BLA
			}
			every_state = { 
				limit = { 
					is_core_of = VOL
				} 
				add_core_of = BLA
			}
		}
	}

	option = {
		name = GOT_Blackfyre_Country.2.a

		ai_chance = {
			factor = 35
		}
		
			BLA = {
				inherit_technology = VOL
				country_event = { id = GOT_Blackfyre_Country.6 }
				create_faction = "The Black Dragons"
			}
			VOL = {
				end_puppet = BLA
			}
			VOL = {
				every_unit_leader = {
					set_nationality = BLA
				}
				transfer_navy = {
					target = BLA
				}
			}
			BLA = {
				change_tag_from = VOL
				inherit_technology = VOL
			}
			BLA = {
				annex_country = {
					target = VOL
					transfer_troops = yes
				}
			}
				
		every_other_country = {
			limit = {
				exists = yes
			}
			news_event = {
				id = GOT_Blackfyre_News.2
				hours = 6
			}
		}
		
	}
	
	option = {
		name = GOT_Blackfyre_Country.2.b

		ai_chance = {
			factor = 65
		}
		
		BLA = {
			inherit_technology = VOL
			country_event = { id = GOT_Blackfyre_Country.6 }
			create_faction = "The Black Dragons"
		}
		
		declare_war_on = {
			target = BLA
			type = annex_everything
		}
		
		every_other_country = {
			limit = {
				exists = yes
			}
			news_event = {
				id = GOT_Blackfyre_News.2
				hours = 6
			}
		}
		
	}
}
