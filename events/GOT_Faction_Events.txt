#Creation of the League of Free Cities
country_event = {
	id = GOT_Faction_Country.1
	title = GOT_Faction_Country.1.t
	desc = GOT_Faction_Country.1.d
	picture = GFX_report_event_military_planning

	is_triggered_only = yes

	option = {
		name = GOT_Faction_Country.1.a
		
		ai_chance = { 
            base = 100
        } 
		
		create_faction = "League of Free Cities"
		
		every_other_country = {
			limit = {
				is_in_faction = no
				NOT = { has_war_with = ROOT }
				OR = {
					tag = BRA
					tag = LOR
					tag = NRV
					tag = QOH
					tag = PEN
					tag = MYR
					tag = TYR
					tag = LYS
					tag = VOL
				}
			}
			country_event = {
				id = GOT_Faction_Country.2
				hours = 6
			}
		}
		
		every_other_country = { 
            limit = { 
                exists = yes 
            } 
            news_event = { 
                id = GOT_Faction_News.1
                hours = 6 
            } 
        }

	}
}

#Creation of the Flayed Men
country_event = {
	id = GOT_Faction_Country.5
	title = GOT_Faction_Country.5.t
	desc = GOT_Faction_Country.5.d
	picture = GFX_report_event_military_planning

	is_triggered_only = yes
	
	immediate = {
		hidden_effect = {
			set_global_flag = GOT_The_Flayed_Men_Flag
		}
	}

	option = {
		name = GOT_Faction_Country.5.a
		
		ai_chance = { 
            base = 75
        } 
			if = {
				limit = {
					has_war_with = DRA
				}
				DRA = {
				   country_event = {
						id = BPD_Better_Peace_Deals_Country.200
						hours = 1
					}
				}
			}
			
			if = {
				limit = {
					has_war_with = STO
				}
				STO = {
				   country_event = {
						id = BPD_Better_Peace_Deals_Country.200
						hours = 1
					}
				}
			}
		
			#If TAR is a faction leader, the faction is disbaned
			if = {
				limit = {
					is_faction_leader = yes
				}
				dismantle_faction = yes
			}

			#If TAR is a faction member, but not the leader, then he just leaves
			if = {
				limit = {
					is_in_faction = yes
				}
				leave_faction = yes
			}
			
			create_faction = "The Flayed Men"
		
		every_other_country = {
			limit = {
				OR = {
					tag = UMB
					tag = KAR
				}
			}
			country_event = {
				id = GOT_Faction_Country.6
				hours = 6
			}
		}
		
		every_other_country = { 
            limit = { 
                exists = yes 
            } 
            news_event = { 
                id = GOT_Faction_News.3
                hours = 6 
            } 
        }

	}
	
	option = {
		name = GOT_Faction_Country.5.b
		
		ai_chance = { 
            base = 25
			modifier = {
				factor = 0
				is_historical_focus_on = yes
			}
        } 
		
		
		add_war_support = -0.05
		
	}
}
