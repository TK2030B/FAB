#Crowned King
country_event = {
	id = GOT_Location_Country.1
	title = GOT_Location_Country.1.t
	desc = GOT_Location_Country.1.d
	picture = GFX_report_event_Red_Keep

	is_triggered_only = yes

	option = {
		name = GOT_Location_Country.1.a

		ai_chance = {
			factor = 10
		}
		
		set_global_flag = GOT_New_Monarch_Chosen_Flag
		
		add_stability = 0.05
		add_political_power = 25
		
		hidden_effect = {
			if = {
				limit = {
					tag = CRO
				}
				every_other_country = {
					limit = {
						has_guaranteed = CRO
					}
					diplomatic_relation = {
						country = CRO
						relation = guarantee
						active = no
					}
				}
			}
		}
		
		if = {
			limit = {
				has_country_leader = {
					name = "Tommen Baratheon"
					ruling_only = yes
				}
			}
			news_event = { 
                id = GOT_Story_News.321 
                hours = 6 
            }
		}
		else_if = {
			limit = {
				GOT_Female_Ruler = yes
			}
			news_event = {
				id = GOT_Location_News.2
				hours = 6
			}
		}
		else = {
			news_event = {
				id = GOT_Location_News.1
				hours = 6
			}
		}

	}
}

		ROOT = { 
			has_war_with = CRO

		}

	}

		set_global_flag = GOT_Fall_of_Kings_Landing_Flag
	}

		name = GOT_Location_News.51.a

			}
		}

		name = GOT_Location_News.51.b

		}
